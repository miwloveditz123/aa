local CONFIG = {
    -- Script URL untuk auto-execute setelah rejoin
    ScriptUrl = "https://raw.githubusercontent.com/miwloveditz123/aa/refs/heads/main/aa",
    
    -- Game ID (PlaceId South Bronx)
    GameId = 10179538382,
    
    -- Auto Rejoin settings
    AutoRejoinEnabled = true,
    RejoinDelay = 3, -- Detik sebelum rejoin
     
    -- Auto Execute settings
    AutoExecuteAfterRejoin = true,
    ExecuteDelay = 15, -- Detik sebelum execute setelah rejoin
     
    -- Auto Play settings
    MaxPlayAttempts = 5,
    PlayRetryDelay = 1,
    
    -- Instant Rejoin On Execute
    InstantRejoinOnExecute = true, -- true = langsung rejoin pas script di-execute
    InstantRejoinOnExecuteDelay = 0, -- delay sebelum langsung rejoin (0 = langsung)
}

local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local LogService = game:GetService("LogService")
local TeleportService = game:GetService("TeleportService")
local VirtualInputManager = cloneref(game:GetService("VirtualInputManager"))
local CoreGui = game:GetService("CoreGui")

local LP = Players.LocalPlayer
local PlayerGui = LP:WaitForChild("PlayerGui")



local function SetPlayButton()
    local LocalPlayer = Players.LocalPlayer
    if not LocalPlayer then return false end
    
    local PlayerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if not PlayerGui then return false end
    
    -- Cari IntroUI
    local intro = PlayerGui:FindFirstChild("IntroUI")
    if not intro then
        -- IntroUI udah ilang = play button udah ke klik
        return true
    end
    
    -- Cari tombol Play
    local playButton = nil
    for _, obj in pairs(intro:GetDescendants()) do
        if obj:IsA("ImageButton") and obj.Name == "Play" then
            playButton = obj
            break
        end
    end
    
    if not playButton then
        return false
    end
    
    -- METODE 1: Fire() event
    pcall(function()
        playButton.MouseButton1Click:Fire()
    end)
    
    -- METODE 2: VirtualInputManager (real mouse click)
    pcall(function()
        local VIM = game:GetService("VirtualInputManager")
        if VIM then
            local pos = playButton.AbsolutePosition
            local size = playButton.AbsoluteSize
            if pos.X > 0 and pos.Y > 0 then
                VIM:SendMouseButtonEvent(pos.X + size.X/2, pos.Y + size.Y/2, 0, true, game, 0)
                task.wait(0.05)
                VIM:SendMouseButtonEvent(pos.X + size.X/2, pos.Y + size.Y/2, 0, false, game, 0)
            end
        end
    end)
    
    -- METODE 3: getconnections (execute semua listener)
    pcall(function()
        local conns = getconnections(playButton.MouseButton1Click)
        if conns then
            for _, conn in pairs(conns) do
                conn:Fire()
            end
        end
    end)
    
    task.wait(1)
    return true
end


local function AutoClickPlay()
    local maxAttempts = CONFIG.MaxPlayAttempts
    local delay = CONFIG.PlayRetryDelay
    local attempts = 0
    local success = false
    
    while attempts < maxAttempts do
        local result = SetPlayButton()
        if result then
            success = true
            break
        end
        attempts = attempts + 1
        task.wait(delay)
    end
    
    return success
end



local function AutoExecute()
    task.wait(3)
    
    if getgenv().FULLY_MS_AFTER_REJOIN then
        print("🔄 [AUTO-EXEC] Starting after rejoin...")
        getgenv().FULLY_MS_AFTER_REJOIN = nil
        
        -- Click Play button
        local clickAttempts = 0
        while clickAttempts < 10 do
            local success = SetPlayButton()
            if success then break end
            clickAttempts = clickAttempts + 1
            task.wait(2)
        end
        
        task.wait(2)
        
        -- ============================================
        -- CALL MAIN FUNCTION DI SINI
        -- ============================================
        -- Ganti dengan function utama lo:
        --   StartFullyFarm()
        --   LoadMainScript()
        --   dll
        
        if _G.StartFullyFarm then
            _G.StartFullyFarm()
        elseif _G.LoadMainScript then
            _G.LoadMainScript()
        end
    else
        print("📌 [AUTO-EXEC] Ready - Use F3 to open UI")
    end
end



local IsDeathTeleport = false

local function MarkDeathTeleport()
    IsDeathTeleport = true
end

_G.MarkDeathTeleport = MarkDeathTeleport

local function DoRejoin()
    if not CONFIG.AutoRejoinEnabled then return end
    
    -- Skip kalau death teleport (legit)
    if IsDeathTeleport then 
        IsDeathTeleport = false
        return 
    end
    
    print("🔄 [REJOIN] Rejoining...")
    
    -- Set flag untuk auto-execute
    getgenv().FULLY_MS_AFTER_REJOIN = true
    
    -- Queue script untuk jalanin setelah rejoin
    queue_on_teleport([[
        queue_on_teleport("task.wait(30)\n" ..
            "getgenv().FULLY_MS_AFTER_REJOIN = true\n" ..
            "getgenv().FULLY_MS_V1 = nil\n" ..
            "loadstring(game:HttpGet(']] .. CONFIG.ScriptUrl .. [['))()")
    ]])
    
    -- Trigger teleport
    TeleportService:Teleport(CONFIG.GameId)
end



local function SetupDeathDetection()
    local function OnCharacterAdded(char)
        local hum = char:WaitForChild("Humanoid", 5)
        if not hum then return end
        
        hum.Died:Connect(function()
            if IsDeathTeleport then
                IsDeathTeleport = false
                return
            end
            
            print("💀 [REJOIN] Natural death detected! Rejoining...")
            task.spawn(DoRejoin)
        end)
    end
    
    local char = LP.Character
    if char then 
        task.spawn(function() OnCharacterAdded(char) end) 
    end
    
    LP.CharacterAdded:Connect(OnCharacterAdded)
end


local function SetupKickDetection()
    task.spawn(function()
        local conn
        conn = LogService.MessageOut:Connect(function(msg, mtype)
            if mtype == Enum.MessageType.MessageError then
                if msg:find("Kicked") or msg:find("Disconnected") or msg:find("Idle") or 
                   msg:find("Removed") or msg:find("Connection lost") or msg:find("Ban") then
                    if conn then conn:Disconnect() end
                    task.spawn(DoRejoin)
                end
            end
        end)
    end)
end

-- ============================================================
--[[
FUNGSI: InstantRejoin()

DESKRIPSI:
  Langsung rejoin tanpa nunggu mati/disconnect/kick
]]

local function InstantRejoin(delay)
    if not CONFIG.AutoRejoinEnabled then return end
    
    local rejoinDelay = delay or 0
    
    if rejoinDelay > 0 then
        print("⚡ [INSTANT REJOIN] Rejoining in " .. rejoinDelay .. " seconds...")
    else
        print("⚡ [INSTANT REJOIN] Rejoining now...")
    end
    
    task.spawn(function()
        if rejoinDelay > 0 then
            task.wait(rejoinDelay)
        end
        
        -- Set flag untuk auto-execute
        getgenv().FULLY_MS_AFTER_REJOIN = true
        getgenv().FULLY_MS_INSTANT_REJOIN_DONE = true
        
        -- Queue script untuk jalanin setelah rejoin
        queue_on_teleport([[
            queue_on_teleport("task.wait(15)\n" ..
                "getgenv().FULLY_MS_AFTER_REJOIN = true\n" ..
                "getgenv().FULLY_MS_INSTANT_REJOIN_DONE = true\n" ..
                "getgenv().FULLY_MS_V1 = nil\n" ..
                "loadstring(game:HttpGet(']] .. CONFIG.ScriptUrl .. [['))()")
        ]])
        
        -- Trigger teleport
        TeleportService:Teleport(CONFIG.GameId)
    end)
end


task.spawn(function()
    -- Step 1: Auto Click Play
    AutoClickPlay()
    
    -- Step 2: Anti AFK

    
    -- Step 3: Death Detection
    SetupDeathDetection()
    
    -- Step 4: Kick Detection
    SetupKickDetection()
    
    -- Step 5: Instant Rejoin On Execute (langsung tanpa pencet tombol)
    if CONFIG.InstantRejoinOnExecute then
        if not getgenv().FULLY_MS_INSTANT_REJOIN_DONE then
            getgenv().FULLY_MS_INSTANT_REJOIN_DONE = true
            if CONFIG.InstantRejoinOnExecuteDelay and CONFIG.InstantRejoinOnExecuteDelay > 0 then
                print("⚡ [AUTO-INSTANT] Langsung rejoin dalam " .. CONFIG.InstantRejoinOnExecuteDelay .. " detik...")
                task.wait(CONFIG.InstantRejoinOnExecuteDelay)
            end
            InstantRejoin(0)
            return
        end
    else
        -- Hanya setup hotkey kalo fitur auto instant dimatiin
        SetupInstantRejoinHotkey()
    end
    
    -- Step 6: Auto Execute
    AutoExecute()
end)
